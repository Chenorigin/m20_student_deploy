cmake_minimum_required(VERSION 3.10)
project(m20_native_inputs)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(drdds REQUIRED CONFIG PATHS /usr/local/lib/cmake/drdds NO_DEFAULT_PATH)
add_executable(native_lidar_input src/native_lidar_input.cpp)
target_include_directories(native_lidar_input PRIVATE /usr/local/include /usr/local/include/dridl
  include)
find_library(DRDDS_LIB drdds PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
find_library(NATIVE_FASTDDS_LIB fastrtps PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
find_library(NATIVE_FASTCDR_LIB fastcdr PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
target_link_libraries(native_lidar_input ${DRDDS_LIB} ${NATIVE_FASTDDS_LIB} ${NATIVE_FASTCDR_LIB} pthread)
set_target_properties(native_lidar_input PROPERTIES BUILD_RPATH /usr/local/lib INSTALL_RPATH /usr/local/lib)
install(TARGETS native_lidar_input DESTINATION lib/${PROJECT_NAME})
ament_package()
