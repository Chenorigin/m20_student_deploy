cmake_minimum_required(VERSION 3.10)
project(m20_student_sdk)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(drdds REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(m20_heightmap_msgs REQUIRED)
set(M20_VENDOR_SDK "/home/user/sdk_deploy/src/M20_sdk_deploy" CACHE PATH "Existing official SDK dependency root")
set(M20_ORT_ROOT "${M20_VENDOR_SDK}/third_party/onnxruntime/arm" CACHE PATH "CPU ONNX Runtime include/lib directory")
if(NOT EXISTS "${M20_ORT_ROOT}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "Set M20_ORT_ROOT to the verified board CPU ONNX Runtime include/lib directory")
endif()
include_directories(plugin vendor/include/types vendor/include/utils vendor/interface/robot
  vendor/interface/user_command vendor/interface/network vendor/state_machine vendor/run_policy
  "${M20_VENDOR_SDK}/third_party/eigen" "${M20_VENDOR_SDK}/third_party/gamepad/include"
  "${M20_ORT_ROOT}/include")
file(GLOB_RECURSE STATE_SRC vendor/state_machine/*.cpp)
file(GLOB_RECURSE INPUT_SRC vendor/interface/user_command/*.cpp)
file(GLOB_RECURSE GAMEPAD_SRC "${M20_VENDOR_SDK}/third_party/gamepad/src/*.cpp")
add_executable(student_control vendor/main.cpp ${STATE_SRC} ${INPUT_SRC} ${GAMEPAD_SRC})
ament_target_dependencies(student_control rclcpp drdds sensor_msgs std_msgs m20_heightmap_msgs)
find_library(ORT_LIB onnxruntime PATHS "${M20_ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(student_control ${ORT_LIB} pthread m rt dl)
set_target_properties(student_control PROPERTIES BUILD_RPATH "${M20_ORT_ROOT}/lib" INSTALL_RPATH "${M20_ORT_ROOT}/lib")
add_executable(student_onnx_probe plugin/onnx_probe.cpp)
add_executable(gamepad_observe plugin/gamepad_observe.cpp)
add_executable(student_heading_probe ../../tests/heading_feedback_probe.cpp)
target_link_libraries(gamepad_observe pthread)
target_link_libraries(student_onnx_probe ${ORT_LIB} pthread)
set_target_properties(student_onnx_probe PROPERTIES BUILD_RPATH "${M20_ORT_ROOT}/lib" INSTALL_RPATH "${M20_ORT_ROOT}/lib")
add_executable(native_cloud_reader plugin/native_cloud_reader.cpp)
target_include_directories(native_cloud_reader PRIVATE /usr/local/include /usr/local/include/dridl)
find_library(DRDDS_LIB drdds PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
find_library(NATIVE_FASTDDS_LIB fastrtps PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
find_library(NATIVE_FASTCDR_LIB fastcdr PATHS /usr/local/lib NO_DEFAULT_PATH REQUIRED)
target_link_libraries(native_cloud_reader ${DRDDS_LIB} ${NATIVE_FASTDDS_LIB} ${NATIVE_FASTCDR_LIB} pthread)
set_target_properties(native_cloud_reader PROPERTIES BUILD_RPATH /usr/local/lib INSTALL_RPATH /usr/local/lib)
install(TARGETS student_control student_onnx_probe student_heading_probe native_cloud_reader gamepad_observe DESTINATION lib/${PROJECT_NAME})
ament_package()
